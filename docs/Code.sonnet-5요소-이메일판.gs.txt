// ===== ① 초기 설정 =====
function setupSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) sheet = ss.insertSheet(SHEET_NAME);
  if (sheet.getLastRow() === 0) {
    sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]);
    sheet.setFrozenRows(1);
  }

  let logSheet = ss.getSheetByName(LOG_SHEET_NAME);
  if (!logSheet) {
    logSheet = ss.insertSheet(LOG_SHEET_NAME);
    logSheet.getRange(1, 1, 1, 3).setValues([["발송일시", "발송대상수", "대상목록"]]);
  }

  // 기존 트리거 정리 후 매일 08:00 트리거 재등록 (중복 등록 방지)
  ScriptApp.getProjectTriggers().forEach((t) => {
    if (t.getHandlerFunction() === "sendTodayEmailTrigger") {
      ScriptApp.deleteTrigger(t);
    }
  });
  ScriptApp.newTrigger("sendTodayEmailTrigger")
    .timeBased()
    .everyDays(1)
    .atHour(8)
    .create();

  SpreadsheetApp.getUi().alert(
    "초기 설정 완료\n\n- '" + SHEET_NAME + "' 시트에 헤더가 준비됐습니다.\n" +
    "- 매일 오전 8시에 자동으로 메일이 발송되도록 트리거가 등록됐습니다.\n" +
    "- 받는 메일 주소는 스크립트 소유자(본인) 계정입니다.\n\n" +
    "다음 단계: [고객관리] > 가짜 데이터 넣기"
  );
}

// ===== 판정 로직 (Validation/rules.js 와 동일) =====
function parseDate_(dateStr) {
  if (!dateStr) return null;
  if (Object.prototype.toString.call(dateStr) === "[object Date]") {
    const d = new Date(dateStr);
    d.setHours(0, 0, 0, 0);
    return d;
  }
  const parts = String(dateStr).split("-");
  if (parts.length !== 3) return null;
  const d = new Date(Number(parts[0]), Number(parts[1]) - 1, Number(parts[2]));
  d.setHours(0, 0, 0, 0);
  return d;
}

function daysBetween_(dateVal, today) {
  const d = parseDate_(dateVal);
  if (!d) return null;
  return Math.round((today.getTime() - d.getTime()) / 86400000);
}

function daysUntilAnnualDate_(mmdd, today) {
  if (!mmdd) return null;
  const parts = String(mmdd).split("-");
  if (parts.length !== 2) return null;
  const mm = Number(parts[0]);
  const dd = Number(parts[1]);
  let target = new Date(today.getFullYear(), mm - 1, dd);
  const t0 = new Date(today.getFullYear(), today.getMonth(), today.getDate());
  if (target < t0) target = new Date(today.getFullYear() + 1, mm - 1, dd);
  return Math.round((target.getTime() - t0.getTime()) / 86400000);
}

const RULES_ = [
  { key: "R1", label: "21일 넘게 연락 안 함", check: (c, today) => {
      const d = daysBetween_(c.lastContactDate, today);
      return d !== null && d > 21;
  }},
  { key: "R2", label: "견적 발송 후 7일 넘게 무응답", check: (c, today) => {
      if (!c.quoteSentDate || c.quoteRepliedDate) return false;
      const d = daysBetween_(c.quoteSentDate, today);
      return d !== null && d > 7;
  }},
  { key: "R3", label: "재구매 시기 1주일 전", check: (c, today) => {
      if (!c.nextPurchaseDueDate) return false;
      const d = daysBetween_(c.nextPurchaseDueDate, today);
      const daysLeft = -d;
      return daysLeft >= 0 && daysLeft <= 7;
  }},
  { key: "R4", label: "미수금 기일 지남", check: (c, today) => {
      if (!c.unpaidAmount || Number(c.unpaidAmount) <= 0 || !c.paymentDueDate) return false;
      const d = daysBetween_(c.paymentDueDate, today);
      return d !== null && d > 0;
  }},
  { key: "R5", label: "생일/기념일 3일 전", check: (c, today) => {
      const b = daysUntilAnnualDate_(c.birthday, today);
      const a = daysUntilAnnualDate_(c.anniversary, today);
      const hitB = b !== null && b >= 0 && b <= 3;
      const hitA = a !== null && a >= 0 && a <= 3;
      return hitB || hitA;
  }},
];

// 시트를 읽어서 오늘 챙길 고객 목록(고객당 1행, 사유 병합)을 반환
function getTodayTargets_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) return [];
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];
  const values = sheet.getRange(2, 1, lastRow - 1, HEADERS.length).getValues();
  const today = new Date();
  today.setHours(0, 0, 0, 0);

  const targets = [];
  values.forEach((row) => {
    const c = {
      id: row[0], name: row[1], phone: row[2],
      lastContactDate: row[3], quoteSentDate: row[4], quoteRepliedDate: row[5],
      nextPurchaseDueDate: row[6], unpaidAmount: row[7], paymentDueDate: row[8],
      birthday: row[9], anniversary: row[10], memo: row[11],
    };
    if (!c.name) return;
    const reasons = [];
    RULES_.forEach((rule) => {
      if (rule.check(c, today)) reasons.push(rule.label);
    });
    if (reasons.length > 0) {
      targets.push({ name: c.name, phone: c.phone, reasons: reasons });
    }
  });
  return targets;
}

// ===== ③ 미리보기 =====
function previewToday() {
  const targets = getTodayTargets_();
  if (targets.length === 0) {
    SpreadsheetApp.getUi().alert("오늘 챙길 고객이 없습니다.");
    return;
  }
  let msg = "오늘(" + Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd") + ") 챙길 고객 " + targets.length + "명\n\n";
  targets.forEach((t, i) => {
    msg += (i + 1) + ". " + t.name + " (" + t.phone + ")\n   → " + t.reasons.join(" + ") + "\n";
  });
  SpreadsheetApp.getUi().alert(msg);
}

// ===== ④ 메일 발송 =====
function buildEmailBody_(targets) {
  const dateStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
  let body = "오늘(" + dateStr + ") 챙겨야 할 고객 " + targets.length + "명입니다.\n\n";
  targets.forEach((t, i) => {
    body += (i + 1) + ". " + t.name + " (" + t.phone + ")\n   사유: " + t.reasons.join(" + ") + "\n\n";
  });
  body += "-- \n이 메일은 구글시트 자동화(고객관리)에서 매일 오전 8시에 자동 발송됩니다.";
  return body;
}

// 발송로그 시트를 보고 "오늘 날짜"로 이미 발송한 기록이 있는지 확인
function hasSentToday_(todayStr) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const logSheet = ss.getSheetByName(LOG_SHEET_NAME);
  if (!logSheet) return false;
  const lastRow = logSheet.getLastRow();
  if (lastRow < 2) return false;
  const dateValues = logSheet.getRange(2, 1, lastRow - 1, 1).getValues();
  const tz = Session.getScriptTimeZone();
  return dateValues.some((row) => {
    const d = row[0];
    if (!d) return false;
    return Utilities.formatDate(new Date(d), tz, "yyyy-MM-dd") === todayStr;
  });
}

function sendTodayEmail_() {
  const dateStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");

  // 같은 날 두 번째 이상 호출이면 중복 발송하지 않고 여기서 끝냄
  // (버튼을 여러 번 눌러도, 자동 트리거와 겹쳐도 하루 1회만 실제 메일이 나감)
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    if (hasSentToday_(dateStr)) {
      return { sent: false, alreadySent: true, count: 0, message: "오늘(" + dateStr + ")은 이미 메일을 보냈습니다. 중복 발송하지 않았습니다." };
    }

    const targets = getTodayTargets_();
    const recipient = Session.getActiveUser().getEmail();

    if (targets.length === 0) {
      return { sent: false, count: 0, message: "오늘은 챙길 고객이 없어 메일을 보내지 않았습니다." };
    }

    const subject = "[고객관리] " + dateStr + " 오늘 챙길 고객 " + targets.length + "명";
    const body = buildEmailBody_(targets);
    MailApp.sendEmail(recipient, subject, body);

    // 발송로그 기록 (이 기록이 다음 호출에서 중복 방지 기준이 됨)
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const logSheet = ss.getSheetByName(LOG_SHEET_NAME);
    if (logSheet) {
      logSheet.appendRow([
        new Date(),
        targets.length,
        targets.map((t) => t.name).join(", "),
      ]);
    }
    return { sent: true, count: targets.length, message: recipient + " 로 메일을 보냈습니다." };
  } finally {
    lock.releaseLock();
  }
}

// 트리거(매일 08:00)에서 호출
function sendTodayEmailTrigger() {
  sendTodayEmail_();
}

// 메뉴 [지금 메일 보내기]에서 호출 (결과를 팝업으로 안내)
function sendTodayEmailManually() {
  const result = sendTodayEmail_();
  SpreadsheetApp.getUi().alert(result.message);
}
